1. Código do Google Apps Script (copie e cole na sua planilha):

Na sua planilha Google, vá em Extensões > Apps Script, apague tudo e cole o código abaixo: 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);

  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "Data/Hora", "Nome Completo", "Endereço", "Lote",
      "Projeto", "CEP", "Cidade", "Telefone", "CPF", "Chave PIX", "Banco"
    ]);
  }

  sheet.appendRow([
    new Date().toLocaleString("pt-BR"),
    data.nome || "",
    data.endereco || "",
    data.lote || "",
    data.projeto || "",
    data.cep || "",
    data.cidade || "",
    data.telefone || "",
    data.cpf || "",
    data.pix || "",
    data.banco || ""
  ]);

  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
